function formatDateHebrew(dateStr){
  const s = String(dateStr ?? '').trim();
  if(!s) return '';
  if(s === '________') return '________';
  // HTML date input returns YYYY-MM-DD
  if(/^\d{4}-\d{2}-\d{2}$/.test(s)){
    const [y,m,d] = s.split('-');
    return `${d}.${m}.${y}`;
  }
  return s;
}

// them. Used when switching INTO a route: the live DOM may still hold the

function applyMultiCardsExact(snap, count) {
  snap = snap || {};
  var next = {};
  for (var i = 2; i <= count; i++) {
    var k = 'm_case_' + i;
    if (Object.prototype.hasOwnProperty.call(snap, k)) next[k] = snap[k];
  }
  window.MULTI_JOINED_CASE_STORE = next;
  for (let i = 1; i <= count; i++) {
    const caseEl = document.getElementById(`m_case_${i}`);
    if (caseEl && !caseEl.readOnly) {
      caseEl.value = Object.prototype.hasOwnProperty.call(snap, `m_case_${i}`) ? snap[`m_case_${i}`] : '';
    }
    MULTI_CARD_SNAPSHOT_FIELDS.forEach(function(prefix){
      const key = `${prefix}${i}`;
      const el = document.getElementById(key);
      if (!el) return;
      const has = Object.prototype.hasOwnProperty.call(snap, key);
      if (el.type === 'checkbox') el.checked = has ? !!snap[key] : false;
      else el.value = has ? snap[key] : '';
    });
    if (typeof updateConsolCaseTitle === 'function') updateConsolCaseTitle(i);
  }
}
